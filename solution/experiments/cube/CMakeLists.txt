cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(bmmms_cube_probe LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 14)
add_executable(bmmms_cube_probe runner.asc)
target_compile_options(bmmms_cube_probe PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
target_link_libraries(bmmms_cube_probe PRIVATE tiling_api register platform unified_dlog dl m graph_base)

option(BMMMS_TRACE_RUNTIME "Trace probe resource lifetime" OFF)
if(BMMMS_TRACE_RUNTIME)
    target_compile_definitions(bmmms_cube_probe PRIVATE BMMMS_TRACE_RUNTIME)
endif()
option(BMMMS_BUILD_SHARED_TEST "Build an independent C++ ABI caller" OFF)
if(BMMMS_BUILD_SHARED_TEST)
    add_library(bmmms_cube_kernel SHARED library.asc)
    target_compile_options(bmmms_cube_kernel PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
    target_link_options(bmmms_cube_kernel PRIVATE -Wl,-z,defs)
    target_link_libraries(bmmms_cube_kernel PRIVATE tiling_api register platform unified_dlog dl m graph_base)
    find_path(BMMMS_ACL_INCLUDE acl/acl.h PATHS "$ENV{ASCEND_HOME_PATH}/include" REQUIRED)
    find_library(BMMMS_ACL_LIBRARY ascendcl PATHS "$ENV{ASCEND_HOME_PATH}/lib64" REQUIRED)
    add_executable(bmmms_cube_shared_runner ../../tests/shared_runner.cpp)
    target_include_directories(bmmms_cube_shared_runner PRIVATE ${BMMMS_ACL_INCLUDE})
    target_link_libraries(bmmms_cube_shared_runner PRIVATE ${BMMMS_ACL_LIBRARY} dl)
endif()
